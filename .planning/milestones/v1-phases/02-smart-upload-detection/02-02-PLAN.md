    Class: class SmartUploadFilter(private val fingerprintRepository: UploadedFingerprintRepository)

    Fields (private):
    - uploadedFingerprints: Set<String> = emptySet()
    - isInitialized: Boolean = false

    Methods:
    1. suspend fun initialize(sessionId: String? = null)
       - If sessionId provided: load from repository.getSessionFingerprints(sessionId)
       - Else: load from repository.getAllFingerprints()
       - Set isInitialized = true

    2. fun initializeWith(fingerprints: Set<String>)
       - Direct initialization without repository call (for CTG baseline)
       - Set uploadedFingerprints = fingerprints
       - Set isInitialized = true

    3. fun filter(items: List<FingerprintSource>): FilterResult
       - require(isInitialized) { "Filter not initialized" }
       - Partition items into newItems and skippedItems based on fingerprint presence
       - Return FilterResult(newItems, skippedItems, skippedItems.size)

    4. fun isNew(fingerprint: String): Boolean
       - Return fingerprint !in uploadedFingerprints

    5. suspend fun recordUpload(entity: UploadedFingerprintEntity)
       - Add fingerprint to in-memory cache
       - Persist via fingerprintRepository.recordUpload(entity)

    6. fun addToCache(fingerprint: String)
       - Add to uploadedFingerprints without persisting

    7. suspend fun refresh(sessionId: String? = null)
       - Reinitialize filter

    Nested types:
    - data class FilterResult(val newItems: List<FingerprintSource>, val skippedItems: List<FingerprintSource>, val skippedCount: Int)
    - interface FingerprintSource { val fingerprint: String }
