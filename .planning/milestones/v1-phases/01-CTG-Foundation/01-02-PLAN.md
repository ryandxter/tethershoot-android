```kotlin
/**
 * Result of comparing two snapshots
 */
data class DiffResult(
    val baselineId: String,                    // ID of baseline snapshot
    val currentId: String,                     // ID of current snapshot
    val added: List<CtgMediaEntry>,           // New files since baseline
    val modified: List<ModifiedEntry>,        // Files that changed size/time
    val deleted: List<CtgMediaEntry>,          // Files removed since baseline
    val unchangedCount: Int,                   // Count of unchanged files
    val computedAt: Long                       // Timestamp of comparison
) {
    val hasChanges: Boolean
        get() = added.isNotEmpty() || modified.isNotEmpty() || deleted.isNotEmpty()
    
    val changeCount: Int
        get() = added.size + modified.size + deleted.size
    
    val summary: String
        get() = buildString {
            if (added.isNotEmpty()) append("+${added.size} ")
            if (modified.isNotEmpty()) append("~${modified.size} ")
            if (deleted.isNotEmpty()) append("-${deleted.size}")
        }.trim()
}

/**
 * Tracks a file that was modified between snapshots
 */
data class ModifiedEntry(
    val baseline: CtgMediaEntry,
    val current: CtgMediaEntry
) {
    val reason: String
        get() = buildString {
            if (baseline.size != current.size) {
                append("Size: ${baseline.displaySize} -> ${current.displaySize}")
            }
            if (baseline.modifiedTime != current.modifiedTime) {
                if (isNotEmpty()) append("; ")
                append("Modified: ${formatTime(baseline.modifiedTime)} -> ${formatTime(current.modifiedTime)}")
            }
        }.trim()
}

/**
 * Compares CTG snapshots to detect media changes.
 * 
 * Immutable - returns new DiffResult on each comparison.
 * Thread-safe - all state is captured in DiffResult.
 */
class CtgStateDiffer {
    
    /**
     * Compare baseline snapshot against current snapshot.
     * Returns diff result describing added, modified, and deleted files.
     */
    fun diff(baseline: CtgMediaSnapshot, current: CtgMediaSnapshot): DiffResult {
        val baselineFingerprints = baseline.entries.keys
        val currentFingerprints = current.entries.keys
        
        // Set difference operations for O(n) complexity
        val addedFps = currentFingerprints - baselineFingerprints
        val deletedFps = baselineFingerprints - currentFingerprints
        val commonFps = baselineFingerprints intersect currentFingerprints
        
        // Check for modifications in common set
        val modifiedFps = commonFps.filter { fp ->
            val baselineEntry = baseline.entries[fp]!!
            val currentEntry = current.entries[fp]!!
            isModified(baselineEntry, currentEntry)
        }.toSet()
        
        val unchangedCount = commonFps.size - modifiedFps.size
        
        return DiffResult(
            baselineId = baseline.id,
            currentId = current.id,
            added = addedFps.map { current.entries[it]!! },
            modified = modifiedFps.map { fp ->
                ModifiedEntry(
                    baseline = baseline.entries[fp]!!,
                    current = current.entries[fp]!!
                )
            },
            deleted = deletedFps.map { baseline.entries[it]!! },
            unchangedCount = unchangedCount,
            computedAt = System.currentTimeMillis()
        )
    }
    
    /**
     * Detect if an entry has been modified (size or timestamp changed)
     */
    private fun isModified(baseline: CtgMediaEntry, current: CtgMediaEntry): Boolean {
        return baseline.size != current.size ||
               baseline.modifiedTime != current.modifiedTime
    }
    
    private fun formatTime(timestamp: Long): String {
        val sdf = java.text.SimpleDateFormat("yyyy-MM-dd HH:mm:ss", java.util.Locale.US)
        return sdf.format(java.util.Date(timestamp))
    }
}
```

**Design notes:**
- Uses Kotlin set operations for O(n) complexity instead of nested loops
- Immutable DiffResult captures all comparison state
- ModifiedEntry tracks both baseline and current for detailed reporting
- Thread-safe: no mutable state in differ

**DO:**
- Use set difference operations (-, intersect) for efficiency
- Compare both size AND modifiedTime for modification detection
- Include unchangedCount for reporting
- Add summary string for logging

**DO NOT:**
- Do not store comparison results in class state
- Do not mutate input snapshots
- Do not add file I/O

```kotlin
/**

suspend fun createBaselineSnapshot(storageId: Int): CtgMediaSnapshot? {
    if (ctgParser == null) {
        Log.w(TAG, "CtgParser not configured, cannot create baseline snapshot")
        return null
    }
    
    val ctgDb = ctgParser.parseCtgFile(storageId) ?: return null
    val snapshot = if (ctgDb.records.isEmpty()) {
        CtgMediaSnapshot.createEmpty(storageId, null)
    } else {
        CtgMediaSnapshot.fromCtgDatabase(ctgDb, storageId, null)
    }
    
    baselineSnapshot = snapshot
    lastEmittedSnapshot = snapshot
    currentSnapshot = snapshot
    
    Log.i(TAG, "Baseline snapshot created: ${snapshot.mediaCount} entries, checksum=${snapshot.checksum}")
    return snapshot
}
```

**Modify checkForChanges() to parse CTG:**
```kotlin
private suspend fun checkForChanges() {
    val storageId = usbHelper.getFirstStorageId()
    if (storageId == 0) return

    val currentMeta = findCtgFile(storageId)

    if (currentMeta != null) {
        val timestampChanged = currentMeta.timestamp > lastKnownTimestamp
        val sizeChanged = currentMeta.size != lastKnownSize

        if (timestampChanged || sizeChanged) {
            Log.i(TAG, ">>> CTG CHANGED! ts=${currentMeta.timestamp} (was ${lastKnownTimestamp}), size=${currentMeta.size} (was ${lastKnownSize})")

            // Emit legacy event for backward compatibility
            val event = CtgChangeEvent(...)
            _ctgChanges.emit(event)

            // NEW: Parse CTG and emit snapshot if parser configured
            if (ctgParser != null) {
                parseAndEmitSnapshot(storageId)
            }

            lastKnownTimestamp = currentMeta.timestamp
            lastKnownSize = currentMeta.size

            _state.value = _state.value.copy(
                changeCount = _state.value.changeCount + 1,
                lastChangeAt = System.currentTimeMillis(),
                lastKnownTimestamp = lastKnownTimestamp,
                lastKnownSize = lastKnownSize,
                statusText = "CTG: Perubahan terdeteksi!"
            )
        }
    }
}

/**
 * Parse CTG file and emit new snapshot if changed
 */
private suspend fun parseAndEmitSnapshot(storageId: Int) {
    val ctgDb = ctgParser?.parseCtgFile(storageId) ?: return
    
    val newSnapshot = if (ctgDb.records.isEmpty()) {
        CtgMediaSnapshot.createEmpty(storageId, null)
    } else {
        CtgMediaSnapshot.fromCtgDatabase(ctgDb, storageId, null)
    }
    
    // Compare with last emitted
    lastEmittedSnapshot?.let { last ->
        val differ = CtgStateDiffer()
        val diff = differ.diff(last, newSnapshot)
        
        if (diff.hasChanges) {
            Log.i(TAG, "CTG content changed: ${diff.summary}")
            _ctgDiffs.emit(diff)
            _ctgSnapshots.emit(newSnapshot)
            lastEmittedSnapshot = newSnapshot
        }
    } ?: run {
        // First snapshot - emit immediately
        _ctgSnapshots.emit(newSnapshot)
        lastEmittedSnapshot = newSnapshot
    }
    
    currentSnapshot = newSnapshot
}
```

```kotlin
package com.reflection.transferin.domain.shoot

import org.junit.Test
import org.junit.Assert.*

class CtgStateDifferTest {
    
    private val differ = CtgStateDiffer()
    
    @Test
    fun `diff detects added files`() {
        val baseline = createSnapshot(listOf(
            createEntry(1, "existing.jpg", 1024)
        ))
        
        val current = createSnapshot(listOf(
            createEntry(1, "existing.jpg", 1024),
            createEntry(2, "new.jpg", 2048)
        ))
        
        val result = differ.diff(baseline, current)
        
        assertTrue(result.hasChanges)
        assertEquals(1, result.added.size)
        assertEquals("new.jpg", result.added[0].filename)
        assertEquals(0, result.modified.size)
        assertEquals(0, result.deleted.size)
    }
    
    @Test
    fun `diff detects deleted files`() {
        val baseline = createSnapshot(listOf(
            createEntry(1, "keep.jpg", 1024),
            createEntry(2, "delete.jpg", 2048)
        ))
        
        val current = createSnapshot(listOf(
            createEntry(1, "keep.jpg", 1024)
        ))
        
        val result = differ.diff(baseline, current)
        
        assertTrue(result.hasChanges)
        assertEquals(0, result.added.size)
        assertEquals(0, result.modified.size)
        assertEquals(1, result.deleted.size)
        assertEquals("delete.jpg", result.deleted[0].filename)
    }
    
    @Test
    fun `diff detects modified files`() {
        val baseline = createSnapshot(listOf(
            createEntry(1, "photo.jpg", 1024, modifiedTime = 1000L)
        ))
        
        val current = createSnapshot(listOf(
            createEntry(1, "photo.jpg", 2048, modifiedTime = 2000L)  // Size and time changed
        ))
        
        val result = differ.diff(baseline, current)
        
        assertTrue(result.hasChanges)
        assertEquals(0, result.added.size)
        assertEquals(1, result.modified.size)
        assertEquals(0, result.deleted.size)
        assertEquals(2048L, result.modified[0].current.size)
        assertTrue(result.modified[0].reason.contains("Size"))
    }
    
    @Test
    fun `diff detects size-only modification`() {
        val baseline = createSnapshot(listOf(
            createEntry(1, "photo.jpg", 1024, modifiedTime = 1000L)
        ))
        
        val current = createSnapshot(listOf(
            createEntry(1, "photo.jpg", 2048, modifiedTime = 1000L)  // Only size changed
        ))
        
        val result = differ.diff(baseline, current)
        
        assertTrue(result.hasChanges)
        assertEquals(1, result.modified.size)
        assertTrue(result.modified[0].reason.contains("Size"))
    }
    
    @Test
    fun `diff handles empty snapshots`() {
        val baseline = CtgMediaSnapshot.createEmpty(1, "camera-1")
        val current = CtgMediaSnapshot.createEmpty(1, "camera-1")
        
        val result = differ.diff(baseline, current)
        
        assertFalse(result.hasChanges)
        assertEquals(0, result.changeCount)
        assertEquals(0, result.unchangedCount)
    }
    
    @Test
    fun `diff counts unchanged correctly`() {
        val baseline = createSnapshot(listOf(
            createEntry(1, "a.jpg", 1024),
            createEntry(2, "b.jpg", 1024),
            createEntry(3, "c.jpg", 1024)
        ))
        
        val current = createSnapshot(listOf(
            createEntry(1, "a.jpg", 1024),  // unchanged
            createEntry(2, "b.jpg", 1024),  // unchanged
            createEntry(3, "c.jpg", 1024),  // unchanged
            createEntry(4, "d.jpg", 1024)   // added
        ))
        
        val result = differ.diff(baseline, current)
        
        assertEquals(3, result.unchangedCount)
        assertEquals(1, result.added.size)
    }
    
    @Test
    fun `summary formats correctly`() {
        val baseline = createSnapshot(listOf(
            createEntry(1, "a.jpg", 1024),
            createEntry(2, "b.jpg", 1024)
        ))
        
        val current = createSnapshot(listOf(
            createEntry(1, "a.jpg", 1024),
            createEntry(3, "c.jpg", 1024),  // added
            createEntry(4, "d.jpg", 1024)   // added
        ))
        
        val result = differ.diff(baseline, current)
        
        assertTrue(result.summary.contains("+2"))
        assertTrue(result.summary.contains("-1"))  // b.jpg deleted
    }
    
    // Helper functions
    private fun createEntry(
        handle: Int,
        filename: String,
        size: Long,
        modifiedTime: Long = System.currentTimeMillis()
    ): CtgMediaEntry {
        return CtgMediaEntry(
            handle = handle,
            filename = filename,
            size = size,
            modifiedTime = modifiedTime,
            capturedTime = null,
            formatCode = 0x3001  // JPEG
        )
    }
    
    private fun createSnapshot(entries: List<CtgMediaEntry>): CtgMediaSnapshot {
        val map = entries.associate { it.fingerprint to it }
        return CtgMediaSnapshot(
            id = "test-${System.nanoTime()}",
            capturedAt = System.currentTimeMillis(),
            storageId = 1,
            cameraId = "test-camera",
            mediaCount = entries.size,
            entries = map,
            checksum = "test-checksum"
        )
    }
}
```
